Firestore Data Model

Collections, documents, and how they connect via userId

/users/{userId}
uid: string
displayName: string
email: string
photoURL: string
isAdmin: boolean
leagueIds: string[]
joinedAt: timestamp
status: active | banned
/picks/{userId}/matches/{matchId}
userId: string ← FK
matchId: string ← FK
homeGoals: number | null
awayGoals: number | null
submittedAt: timestamp
lockedAt: timestamp
points: number | null
pointsBreakdown: {
  exact: bool,
  correctGap: bool,
  correctWinner: bool
}
/matches/{matchId}
apiId: number ← api-football id
homeTeam: { code, name }
awayTeam: { code, name }
kickoff: timestamp
lockAt: timestamp ← kickoff−15m
stage: group|r16|qf|sf|final
stageLabel: string
status: upcoming|live|finished
score: { home: n, away: n } | null
liveMinute: number | null
locked: boolean
/leaderboard/{leagueId}/entries/{userId}
userId: string ← FK
displayName: string
totalPoints: number
exactScores: number
rank: number
prevRank: number
lastUpdated: timestamp
/leagues/{leagueId}
name: string
code: string ← join code
color: string
memberIds: string[]
createdBy: string ← admin uid
createdAt: timestamp
/bonusPicks/{userId}
userId: string ← FK
topScorer: {
  playerId: number,
  playerName: string,
  submittedAt: timestamp
}
worldCupWinner: {
  teamCode: string,
  submittedAt: timestamp
}
/settings/scoring (single document)
stageMultipliers: {
  group: 1,
  r16: 2,
  qf: 3,
  sf: 4,
  final: 5
}
basePoints: {
  exactScore: 3,
  correctGap: 2,
  correctWinner: 1
}
bonusPoints: {
  topScorer: 25,
  worldCupWinner: 50
}
/invites/{token}
email: string
invitedBy: string ← admin uid
leagueId: string | null
used: boolean
expiresAt: timestamp
createdAt: timestamp
